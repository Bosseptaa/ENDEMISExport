// Объект: MaterialExpressionCustom_4 ключ: Code
// Входы: uv, inTexture, TexInvR
// Выход: CMOT_Float2
// Описание: Old

struct Functions
{
  float Sig(float c)
  {
      return (c > 0.0) ? 1.0 : 0.0;
  }

 float interp(float SC, float SL)
 {
	if (SC >= 0.0 && SL < 0.0)
	{
		return (SC / (SC - SL));
	} else
	{
		return 0.0;
	}
 }
};

 
float SC = Texture2DSample(inTexture, inTextureSampler, uv * TexInvR).x;
float SL = Texture2DSample(inTexture, inTextureSampler, (uv+float2(-1,0)) * TexInvR).x;
float SR = Texture2DSample(inTexture, inTextureSampler, (uv+float2(1,0)) * TexInvR).x;
float ST = Texture2DSample(inTexture, inTextureSampler, (uv+float2(0,1)) * TexInvR).x;
float SB = Texture2DSample(inTexture, inTextureSampler, (uv+float2(0,-1)) * TexInvR).x;

Functions f;
float4 neighbors = float4(f.Sig(SL), f.Sig(SR), f.Sig(ST), f.Sig(SB));
float center = f.Sig(SC);
 
float test = 1.0; // = dot(1.0-neighbors, float4(center, center, center,center));
if (SC > 0.0)
{
	if (SL <= 0.0 || SR <= 0.0 || ST <= 0.0 || SB <= 0.0)
		test = 0.0;
} else
{
	if (SL > 0.0 || SR > 0.0  || ST > 0.0  || SB > 0.0)
		test = 0.0;
}

if(dot(1.0-neighbors, float4(center, center, center,center)) >= 0.5)
{
	float2 offset = float2(0.0, 0.0);
	
	offset.x -= f.interp(SC, SL);	
 	offset.x += f.interp(SC, SR);
	offset.y -= f.interp(SC, SB);	
 	offset.y += f.interp(SC, ST);
	
	if	(SC > 0.0)
	{
		return -float2(uv + offset);	
	}
	else
	{
		return float2(uv - offset);
	}
	
}	 else
{
	if(SC > 0.0)
	{
		return -float2(65535, 65535);
	}
	else
	{
 		return float2(65535,65535);
	}
}
