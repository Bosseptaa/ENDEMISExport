float henyey_greenstein_phase_func(float mu, float g)
{
	#define PI 3.14159265359
	return
						(1. - g*g)
	/ //---------------------------------------------
		((4. * PI) * pow(1. + g*g - 2.*g*mu, 1.5));
}

Functions f;
float RayStepSize = (tmax - tmin) / float(raysteps);
float CurrentDepth = tmin + dither * RayStepSize;
float stepdensity = density * RayStepSize;

float Transmittance = 1.;
float3 Illumination = float3(0.0 ,0.0, 0.0);

for(int i=0; i < raysteps; i++)
{	
	float3 samplePosition = origin + dir * CurrentDepth;
	CurrentDepth += RayStepSize;
	
	float3 SampleToLightDirection = normalize(lightpos - samplePosition);

	float phase = f.henyey_greenstein_phase_func( dot(SampleToLightDirection, dir) , gp);
	
	
	float3 sampleLocal = samplePosition / lightSize;
	float3 tex = Texture2DSample(tex_in, tex_inSampler, sampleLocal.yz / (sampleLocal.x) -0.5);
 
	float attenuation = (1.0 - sampleLocal.x);
	float LightShadow = tex.x * attenuation * attenuation;
 
	
	//float LightShadow = samplerColor.r; //f.getsuncolor(samplePosition, SampleToLightDirection, lightpos, lightnormal);
	float InOutScattering = (indirectLight + directLight * phase) * LightShadow * scatteringfactor;

	Transmittance *= exp(-stepdensity * extincting);
	Illumination += stepdensity * Transmittance * InOutScattering;	
}

return Illumination * Transmittance;
