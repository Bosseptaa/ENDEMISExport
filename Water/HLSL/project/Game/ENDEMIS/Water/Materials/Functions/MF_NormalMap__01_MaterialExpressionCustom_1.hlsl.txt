float x = offset.x;
float y = offset.y;

float n0 = Tex.SampleLevel(TexSampler, uv + float2(-x,-y),0).x;
float n1 = Tex.SampleLevel(TexSampler, uv + float2( 0, -y),0).x;
float n2 = Tex.SampleLevel(TexSampler, uv + float2(+x,-y),0).x;
float n3 = Tex.SampleLevel(TexSampler, uv + float2(-x,  0),0).x;
float n4 = Tex.SampleLevel(TexSampler, uv + float2( 0, 0),0).x;
float n5 = Tex.SampleLevel(TexSampler, uv + float2(+x,  0),0).x;
float n6 = Tex.SampleLevel(TexSampler, uv + float2(-x,+y),0).x;
float n7 = Tex.SampleLevel(TexSampler, uv + float2( 0, +y),0).x;
float n8 = Tex.SampleLevel(TexSampler, uv + float2(+x,+y),0).x;

float sobel_edge_h = n2 + (2.0*n5) + n8 - (n0 + (2.0*n3) + n6);
float sobel_edge_v = n0 + (2.0*n1) + n2 - (n6 + (2.0*n7) + n8);
float sobel = sqrt((sobel_edge_h * sobel_edge_h) + (sobel_edge_v * sobel_edge_v));

return sobel;
