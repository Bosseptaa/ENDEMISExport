// Объект: MaterialExpressionCustom_1 ключ: Code
// Входы: WorldPosition, MaxThickness, MaxIterations, Reflection
// Выход: CMOT_Float2
// Описание: SSR

float2 result = float2(0.0,0.0);
float3 rayPos = WorldPosition;
float prevDepth;
for (int n = 0; n < MaxIterations; n++)
{
	float4 vpPos = mul(float4(rayPos, 1.0f), LWCToFloat(ResolvedView.WorldToClip));
	float2 rayUv = (vpPos.xy / vpPos.w) * float2(0.5,-0.5) + 0.5;
 
    float rayDepth = vpPos.w;
    float gbufferDepth = CalcSceneDepth(float2(ViewportUVToBufferUV(rayUv))).x;
   	Distance = gbufferDepth; //min(gbufferDepth,3000.0);
		 
	float depthDiff = rayDepth - gbufferDepth;
	if (depthDiff > 0 && depthDiff < MaxThickness)
	{  
		return lerp(result,rayUv,(0 - prevDepth)/ (depthDiff - prevDepth));
	}
	prevDepth = depthDiff;
	result = rayUv;
	rayPos += Reflection;
}
//Distance = 100000.0;
return result;
