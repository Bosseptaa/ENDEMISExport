struct Functions
{
float4 saturate(float4 v) {
  return clamp(v, 0.0, 1.0);
}

float gaussian(float v, float twoSigmaSq) {
  return exp(-(v * v) / twoSigmaSq);  
}

float4 gaussian4(float4 v, float twoSigmaSq) {
  return exp(-(v * v) / twoSigmaSq);
}

float4 truncatedGaussian(float4 v, float twoSigmaSq, float trunc) {
  return saturate(gaussian4(v, twoSigmaSq) - trunc); 
}
};

  float4 xDist1 = xDist0 + one;
  float4 yDist0 = -a.yyyy + float4(-3.0, -1.0, 1.0, 3.0);
  float4 yDist1 = yDist0 + one;

  float4 gx0 = f.truncatedGaussian(xDist0, twoSigmaSq, trunc);
  float4 gx1 = f.truncatedGaussian(xDist1, twoSigmaSq, trunc);
  float4 gy0 = f.truncatedGaussian(yDist0, twoSigmaSq, trunc);
